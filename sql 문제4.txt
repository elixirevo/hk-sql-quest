SELECT
  ADMINIST_ZONE_NO,
  ADMINIST_ZONE_NM,
  십대인구 / NULLIF(전체인구, 0) AS 십대비율,
  육십대인구 / NULLIF(전체인구, 0) AS 육십대비율
FROM (
  SELECT
    ADMINIST_ZONE_NO,
    ADMINIST_ZONE_NM,
    SUM(CASE WHEN AGRDE_SE_CD = '010' THEN POPLTN_CNT END) AS 십대인구,
    SUM(CASE WHEN AGRDE_SE_CD = '060' THEN POPLTN_CNT END) AS 육십대인구,
    SUM(POPLTN_CNT) AS 전체인구
  FROM TB_POPLTN
  WHERE STD_MT = '202304'
    AND POPLTN_SE_CD = 'T'
    AND ADMINIST_ZONE_NO LIKE '________00'
    AND ADMINIST_ZONE_NO NOT LIKE '__00000000'
    AND ADMINIST_ZONE_NO NOT LIKE '_____00000'
  GROUP BY ADMINIST_ZONE_NO, ADMINIST_ZONE_NM
) dd
WHERE (십대인구 / NULLIF(전체인구, 0)) >= 3 * (육십대인구 / NULLIF(전체인구, 0))
ORDER BY 십대비율 DESC;
